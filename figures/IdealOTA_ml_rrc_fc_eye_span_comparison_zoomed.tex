\documentclass[multi=minipage]{standalone}
\usepackage{amsmath}
\usepackage{pgfplots}
\usepackage{luacode}
\pgfplotsset{compat=1.18}

\begin{document}

    \begin{luacode*}
        -- Define the sps range matching your 9 folder names
        local sps_list = {
            {folder = "4sps", val = "4", label = "4"},
            {folder = "9sps", val = "9", label = "9"},
            {folder = "16sps", val = "16", label = "16"},
            {folder = "25sps", val = "25", label = "25"},
            {folder = "36sps", val = "36", label = "36"},
            {folder = "49sps", val = "49", label = "49"},
            {folder = "64sps", val = "64", label = "64"},
            {folder = "81sps", val = "81", label = "81"},
        }

        -- Set the fixed roll-off factor to track
        local alpha_str = "0.70" 

        -- Base path to the directories
        local base_path = "../data/ideal/eye/"

        for i, item in ipairs(sps_list) do

            -- Construct relative filepaths for both spans
            local file_span4 = string.format(
                "%s%s/ml_rrc_fc_eye_%ssps_span4_alpha%s.txt", 
                base_path, item.folder, item.val, alpha_str
            )
            local file_span12 = string.format(
                "%s%s/ml_rrc_fc_eye_%ssps_span12_alpha%s.txt", 
                base_path, item.folder, item.val, alpha_str
            )

            tex.print("\\begin{minipage}{18cm}")

            -- ==================== LEFT PLOT: SPAN 4 ====================
            tex.print("\\begin{tikzpicture}")
            tex.print("\\begin{axis}[")
            tex.print("    title={\\textsf{Span = 4 ($sps = " .. item.label .. "$)}},")
            tex.print("    xlabel={Time ($T_{sym}$)},")
            tex.print("    ylabel={Amplitude},")
            tex.print("    xmin=0, xmax=3,")
            -- Crucial: Fix ymin and ymax globally to eliminate visual axis jumping
            tex.print("    ymin=540, ymax=660,")
            tex.print("    grid=major,")
            tex.print("    width=0.49\\linewidth,")
            tex.print("    height=0.42\\linewidth,")
            tex.print("    ticklabel style={font=\\tiny},")
            tex.print("    label style={font=\\scriptsize},")
            tex.print("    title style={font=\\small},")
            tex.print("    every axis plot/.append style={")
            tex.print("        mark=none,")
            tex.print("        color=blue,")
            tex.print("        line width=0.35pt,")
            tex.print("        opacity=0.25")
            tex.print("    }")
            tex.print("]")
            tex.print("\\addplot table[x=Time, y=Amplitude] {" .. file_span4 .. "};")
            tex.print("\\end{axis}")
            tex.print("\\end{tikzpicture}%")
            tex.print("\\hfill%")

            -- ==================== RIGHT PLOT: SPAN 12 ====================
            tex.print("\\begin{tikzpicture}")
            tex.print("\\begin{axis}[")
            tex.print("    title={\\textsf{Span = 12 ($sps = " .. item.label .. "$)}},")
            tex.print("    xlabel={Time ($T_{sym}$)},")
            tex.print("    ylabel={Amplitude},")
            tex.print("    xmin=0, xmax=3,")
            tex.print("    ymin=540, ymax=660,")
            tex.print("    grid=major,")
            tex.print("    width=0.49\\linewidth,")
            tex.print("    height=0.42\\linewidth,")
            tex.print("    ticklabel style={font=\\tiny},")
            tex.print("    label style={font=\\scriptsize},")
            tex.print("    title style={font=\\small},")
            tex.print("    every axis plot/.append style={")
            tex.print("        mark=none,")
            tex.print("        color=red!80!black,")
            tex.print("        line width=0.35pt,")
            tex.print("        opacity=0.25")
            tex.print("    }")
            tex.print("]")
            tex.print("\\addplot table[x=Time, y=Amplitude] {" .. file_span12 .. "};")
            tex.print("\\end{axis}")
            tex.print("\\end{tikzpicture}")

            tex.print("\\end{minipage}")
        end

    \end{luacode*}
    
\end{document}
