\documentclass[multi=tikzpicture]{standalone}
\usepackage{amsmath}
\usepackage{pgfplots}
\usepackage{luacode}
\pgfplotsset{compat=1.18}

\begin{document}

    \begin{luacode*}
        local snr_list = {
            {val = "00", label = "0"},
            {val = "05", label = "5"},
            {val = "10", label = "10"},
            {val = "15", label = "15"},
            {val = "20", label = "20"}
        }

        local base_dir = "../data/ideal/rrc/eye/"

        for i, item in ipairs(snr_list) do
            local file_path = string.format(
                "%sgr_rrc_fc_eye_12sps_span12_alpha0.70_snr%sdB.dat",
                base_dir, item.val
            )

            tex.print("\\begin{tikzpicture}")
            tex.print("\\begin{axis}[")
            tex.print("    title={\\textsf{GNU Radio Matched RRC Eye ($\\text{SNR} = " .. item.label .. "\\text{ dB}$)}},")
            tex.print("    xlabel={Time ($T_{sym}$)},")
            tex.print("    ylabel={Amplitude},")
            tex.print("    xmin=0, xmax=3,")
            tex.print("    ymin=10, ymax=45,")
            tex.print("    grid=major,")
            tex.print("    width=9cm,")
            tex.print("    height=7.2cm,")
            tex.print("    ticklabel style={font=\\tiny},")
            tex.print("    label style={font=\\scriptsize},")
            tex.print("    title style={font=\\small}")
            tex.print("]")
            tex.print("\\addplot[")
            tex.print("    no markers,")
            tex.print("    color=red!80!black,")
            tex.print("    line width=0.25pt,")
            tex.print("    opacity=0.35,")
            tex.print("    line cap=round,")
            tex.print("    line join=round")
            tex.print("] table[x=Time, y=Amplitude] {" .. file_path .. "};")
            tex.print("\\end{axis}")
            tex.print("\\end{tikzpicture}")
        end
    \end{luacode*}

\end{document}
